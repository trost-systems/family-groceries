import { type Page } from '@playwright/test';
import { expect, test } from './fixtures';
import { typeInto as newTypeInto } from './helpers';

// Probe for job 12b, not committed: is text inserted after DOM focus but before Flutter's next
// frame lost? Frames are held back by queueing requestAnimationFrame callbacks.

async function holdFrames(page: Page) {
  await page.evaluate(() => {
    const w = window as unknown as Record<string, unknown>;
    const raf = window.requestAnimationFrame.bind(window);
    const held: FrameRequestCallback[] = [];
    w.__held = held;
    window.requestAnimationFrame = (cb) => (held.push(cb), 0);
    w.__release = () => {
      window.requestAnimationFrame = raf;
      held.splice(0).forEach((cb) => raf(cb));
    };
  });
}
const release = (page: Page) => page.evaluate(() => (window as unknown as { __release: () => void }).__release());
const heldCount = (page: Page) => page.evaluate(() => (window as unknown as { __held: unknown[] }).__held.length);
const frames = (page: Page) => page.evaluate(() => new Promise((r) => requestAnimationFrame(() => requestAnimationFrame(r))));

for (const typeBeforeFrame of [true, false]) {
  test(`insert ${typeBeforeFrame ? 'before' : 'after'} the frame that follows focus`, async ({ page }) => {
    await page.goto('/');
    const field = page.getByRole('textbox', { name: 'Passwort' });
    await expect(field).toBeVisible();
    await frames(page);
    await holdFrames(page);
    await field.click();
    await expect(field).toBeFocused();
    if (!typeBeforeFrame) {
      await release(page);
      await frames(page);
    }
    await page.keyboard.insertText('abcdefgh');
    const domRightAfter = await field.inputValue();
    const held = typeBeforeFrame ? await heldCount(page) : 0;
    if (typeBeforeFrame) await release(page);
    await frames(page);
    await page.waitForTimeout(500);
    const domAfterFrames = await field.inputValue();
    // What the app holds: Registrieren validates the password field (>= 6 characters).
    await page.getByRole('button', { name: 'Registrieren' }).click();
    const tooShort = await page.getByText(/mindestens|zu kurz/i).count();
    console.log(`PROBE insert ${typeBeforeFrame ? 'before' : 'after'} frame: held frames=${held}, DOM right after=${JSON.stringify(domRightAfter)}, DOM after frames=${JSON.stringify(domAfterFrames)}, app says too short=${tooShort > 0}`);
  });
}

async function oldTypeInto(field: import('@playwright/test').Locator, text: string): Promise<void> {
  await expect(async () => {
    await field.click();
    await expect(field).toBeFocused();
    await field.page().keyboard.press('ControlOrMeta+a');
    await field.page().keyboard.insertText(text);
    await expect(field).toHaveValue(text, { timeout: 2_000 });
  }).toPass({ timeout: 20_000 });
}

for (const [name, type] of [['old', oldTypeInto], ['new', newTypeInto]] as const) {
  test(`typeInto ${name} while the frame after focus is 1.5 s late`, async ({ page }) => {
    await page.goto('/');
    const field = page.getByRole('textbox', { name: 'Passwort' });
    await expect(field).toBeVisible();
    await frames(page);
    await holdFrames(page);
    await page.evaluate(() => setTimeout((window as unknown as { __release: () => void }).__release, 1_500));
    await type(field, 'abcdefgh');
    await page.waitForTimeout(2_000);
    await page.getByRole('button', { name: 'Registrieren' }).click();
    const tooShort = await page.getByText(/mindestens/i).count();
    console.log(`PROBE typeInto ${name}, frame 1.5 s late: DOM=${JSON.stringify(await field.inputValue())}, app says too short=${tooShort > 0}`);
  });
}
